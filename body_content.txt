function addCEvents(cv){
  // Maps a client (screen) coordinate to actual canvas pixel coordinate,
  // accounting for CSS scaling (canvas display size != internal pixel size on mobile)
  function toCanvasXY(clientX, clientY){
    const rc=cv.getBoundingClientRect();
    const scaleX=cv.width/rc.width;
    const scaleY=cv.height/rc.height;
    return { x:(clientX-rc.left)*scaleX, y:(clientY-rc.top)*scaleY };
  }

  cv.onmousedown=e=>{
    S.drawing=true;
    const p=toCanvasXY(e.clientX,e.clientY);
    S.sx=p.x; S.sy=p.y;
  };
  cv.onmousemove=e=>{
    if(!S.drawing)return;
    const p=toCanvasXY(e.clientX,e.clientY);
    drawCCanvas();
    const ctx=cv.getContext('2d');
    ctx.strokeStyle='#00d4a0'; ctx.lineWidth=2; ctx.setLineDash([6,3]);
    ctx.strokeRect(S.sx,S.sy,p.x-S.sx,p.y-S.sy); ctx.setLineDash([]);
  };
  cv.onmouseup=e=>{
    if(!S.drawing)return; S.drawing=false;
    const p=toCanvasXY(e.clientX,e.clientY);
    finishCRegion(p.x,p.y);
  };
  cv.onmouseleave=()=>{ if(S.drawing){ S.drawing=false; drawCCanvas(); } };

  function finishCRegion(x,y){
    const rx=Math.min(S.sx,x), ry=Math.min(S.sy,y);
    const rw=Math.abs(x-S.sx), rh=Math.abs(y-S.sy);
    if(rw<5||rh<5)return;
    const id=S.cRegions.length+1;
    const lbl=document.getElementById('c-name').value+' R'+id;
    S.cRegions.push({id,x:rx,y:ry,w:rw,h:rh,label:lbl});
    drawCCanvas();
    measureRegion(S.cRegions[S.cRegions.length-1]);
  }

  // ── Touch: single-finger drag draws a region. Two-finger pinch zooms (handled separately, drawing is suppressed). ──
  let touchDrawing=false, pinching=false;
  cv.ontouchstart=e=>{
    if(e.touches.length>=2){ pinching=true; touchDrawing=false; S.drawing=false; return; }
    pinching=false;
    e.preventDefault();
    const t=e.touches[0];
    const p=toCanvasXY(t.clientX,t.clientY);
    touchDrawing=true; S.drawing=true; S.sx=p.x; S.sy=p.y;
  };
  cv.ontouchmove=e=>{
    if(pinching||e.touches.length>=2){ pinching=true; touchDrawing=false; S.drawing=false; return; }
    if(!touchDrawing)return;
    e.preventDefault();
    const t=e.touches[0];
    const p=toCanvasXY(t.clientX,t.clientY);
    drawCCanvas();
    const ctx=cv.getContext('2d');
    ctx.strokeStyle='#00d4a0'; ctx.lineWidth=3; ctx.setLineDash([8,4]);
    ctx.strokeRect(S.sx,S.sy,p.x-S.sx,p.y-S.sy); ctx.setLineDash([]);
  };
  cv.ontouchend=e=>{
    if(pinching){ pinching=false; return; }
    if(!touchDrawing)return;
    touchDrawing=false; S.drawing=false;
    e.preventDefault();
    const t=e.changedTouches[0];
    const p=toCanvasXY(t.clientX,t.clientY);
    finishCRegion(p.x,p.y);
  };

  setupPinchZoom(cv, 'c');
}

// ── PINCH-TO-ZOOM for touch devices ─────────────────────────────
// Applies a CSS transform-based zoom to the canvas wrapper so users on
// phones can zoom into small reaction spots before drawing a region box.
function setupPinchZoom(cv, prefix){
  const wrap = cv.closest('.cbox');
  let scale=1, lastDist=0, originX=0, originY=0;
  wrap.style.transformOrigin='0 0';
  wrap.addEventListener('touchstart', e=>{
    if(e.touches.length===2){
      lastDist=Math.hypot(
        e.touches[0].clientX-e.touches[1].clientX,
        e.touches[0].clientY-e.touches[1].clientY
      );
    }
  }, {passive:true});
  wrap.addEventListener('touchmove', e=>{
    if(e.touches.length===2){
      e.preventDefault();
      const dist=Math.hypot(
        e.touches[0].clientX-e.touches[1].clientX,
        e.touches[0].clientY-e.touches[1].clientY
      );
      if(lastDist>0){
        scale*=dist/lastDist;
        scale=Math.min(4,Math.max(1,scale));
        cv.style.transform=`scale(${scale})`;
        cv.style.transformOrigin='0 0';
      }
      lastDist=dist;
    }
  }, {passive:false});
  wrap.addEventListener('touchend', e=>{
    if(e.touches.length<2) lastDist=0;
  }, {passive:true});
  // expose a reset so "Clear Regions" can also reset zoom
  cv.dataset.zoomReset='1';
  wrap._resetZoom=()=>{ scale=1; cv.style.transform='scale(1)'; };
}

// ── COLOUR MATHS ──────────────────────────────────────────────
function getPixels(cv,x,y,w,h){
  return cv.getContext('2d').getImageData(
    Math.round(x),Math.round(y),Math.max(1,Math.round(w)),Math.max(1,Math.round(h))
  ).data;
}
function meanRGB(px){
  let r=0,g=0,b=0,n=0;
  for(let i=0;i<px.length;i+=4){r+=px[i];g+=px[i+1];b+=px[i+2];n++;}
  return{R:Math.round(r/n),G:Math.round(g/n),B:Math.round(b/n)};
}
function toHSV(r,g,b){
  r/=255;g/=255;b/=255;
  const M=Math.max(r,g,b),m=Math.min(r,g,b),d=M-m;
  let h=0,s=M?d/M:0,v=M;
  if(d){
    if(M===r)h=((g-b)/d)%6;
    else if(M===g)h=(b-r)/d+2;
    else h=(r-g)/d+4;
    h=Math.round(h*60);if(h<0)h+=360;
  }
  return{H:h,S:+(s*100).toFixed(1),V:+(v*100).toFixed(1)};
}
function toLAB(r,g,b){
  let R=r/255,G=g/255,B=b/255;
  const lin=v=>v>.04045?Math.pow((v+.055)/1.055,2.4):v/12.92;
  R=lin(R);G=lin(G);B=lin(B);
  let X=R*.4124+G*.3576+B*.1805,Y=R*.2126+G*.7152+B*.0722,Z=R*.0193+G*.1192+B*.9505;
  X/=.95047;Y/=1;Z/=1.08883;
  const f=t=>t>.008856?Math.cbrt(t):7.787*t+16/116;
  return{L:+(116*f(Y)-16).toFixed(1),A:+(500*(f(X)-f(Y))).toFixed(1),B:+(200*(f(Y)-f(Z))).toFixed(1)};
}
function toOD(rgb,blank={R:240,G:240,B:240}){
  const I=(rgb.R+rgb.G+rgb.B)/3, I0=(blank.R+blank.G+blank.B)/3;
  return+(Math.max(0,Math.log10(I0/Math.max(1,I)))).toFixed(4);
}
function dE(l1,l2){
  return+Math.sqrt((l1.L-l2.L)**2+(l1.A-l2.A)**2+(l1.B-l2.B)**2).toFixed(2);
}
function concFromOD(od){
  if(!S.calCurve)return null;
  return+Math.max(0,(od-S.calCurve.intercept)/S.calCurve.slope).toFixed(3);
}
function classify(hb,grp){
  const t={child:11,'teen-m':13,'teen-f':12,'adult-m':13,'adult-f':12,pregnant:11,'post-m':12,elder:12};
  const thr=t[grp]??12;
  if(hb>=thr) return{l:'Normal',c:'bg'};
  if(hb>=10)  return{l:'Mild Anaemia',c:'by'};
  if(hb>=7)   return{l:'Moderate Anaemia',c:'bo'};
  return{l:'Severe Anaemia',c:'br'};
}

function measureRegion(r){
  const cv=document.getElementById('c-cv');
  const px=getPixels(cv,r.x,r.y,r.w,r.h);
  const rgb=meanRGB(px);
  const hsv=toHSV(rgb.R,rgb.G,rgb.B);
  const lab=toLAB(rgb.R,rgb.G,rgb.B);
  const blank=S.cImages[1]?getBlankRGB():null;
  const od=toOD(rgb,blank);
  const de=dE(lab,{L:96,A:0,B:2});
  const conc=concFromOD(od);
  Object.assign(r,{rgb,hsv,lab,od,dE:de,conc});
  const idx=S.cRegions.findIndex(x=>x.id===r.id);
  if(idx>=0)S.cRegions[idx]=r;
  renderCResults();
  updateCCharts(r,px);
}

}

function buildCurve(){
  const cc=[...document.querySelectorAll('.cc')];
  const ca=[...document.querySelectorAll('.ca')];
  const pts=[];
  cc.forEach((ci,i)=>{
    const x=parseFloat(ci.value),y=parseFloat(ca[i].value);
    if(!isNaN(x)&&!isNaN(y))pts.push({x,y});
  });
  if(pts.length<3){toast('Need at least 3 data points');return;}
  const n=pts.length;
  const sx=pts.reduce((a,p)=>a+p.x,0), sy=pts.reduce((a,p)=>a+p.y,0);
  const sxy=pts.reduce((a,p)=>a+p.x*p.y,0), sx2=pts.reduce((a,p)=>a+p.x*p.x,0);
  const slope=(n*sxy-sx*sy)/(n*sx2-sx*sx);
  const intercept=(sy-slope*sx)/n;
  const ym=sy/n;
  const r2=1-pts.reduce((a,p)=>a+(p.y-(slope*p.x+intercept))**2,0)/pts.reduce((a,p)=>a+(p.y-ym)**2,0);
  S.calCurve={slope,intercept,r2,pts};
  document.getElementById('cal-eq').innerHTML=
    `✓ <b>Absorbance = ${slope.toFixed(5)} × Conc + ${intercept.toFixed(5)}</b><br>
     R² = <b>${r2.toFixed(4)}</b> ${r2>.99?'✅ Excellent':r2>.95?'✓ Good':'⚠ Check data — values should increase smoothly'}`;
  S.cRegions.forEach(r=>{if(r.od!=null)r.conc=concFromOD(r.od);});
  renderCResults();
  drawCurveChart();
  toast('✓ Calibration curve built');
}

// Healthy reference values for Delta-E and flag detection
const HEALTHY = {
  plasma : { rgb:[228,212,160], lab:{L:86,A:2,B:22} },
  buffy  : { rgb:[210,205,190], lab:{L:82,A:1,B:10} },

// Pallor index: how pale are RBCs vs healthy (0=normal,100=severely pale)
function pallor(labL){ return +Math.min(100,Math.max(0,(labL-38)/0.62)).toFixed(1); }

// Icterus (yellow plasma): LAB B > 32 is clinically elevated bilirubin proxy
function icterusFlag(lab){ return lab.B>32; }

// Lipemia (milky plasma): high lightness + very low saturation
function lipemiaFlag(hsv,lab){ return lab.L>88 && hsv.S<12; }

// Buffy coat visibility score (0–3)
function buffyScore(hpct, sat){
  if(hpct<1)  return {n:0,l:'Not visible'};
  if(hpct<2)  return {n:1,l:'Faintly visible'};
  if(hpct<5)  return {n:2,l:'Clearly visible'};
  return            {n:3,l:'Abnormally thick — possible infection/inflammation'};
}

function addBEvents(cv){
  let bdraw=false,bsx,bsy;
  function toXY(clientX,clientY){
    const rc=cv.getBoundingClientRect();
    return{x:(clientX-rc.left)*(cv.width/rc.width),y:(clientY-rc.top)*(cv.height/rc.height)};
  }
  cv.onmousedown=e=>{ bdraw=true; const p=toXY(e.clientX,e.clientY); bsx=p.x; bsy=p.y; };
  cv.onmousemove=e=>{
    if(!bdraw)return;
    const p=toXY(e.clientX,e.clientY);
    drawBCanvas();
    const lc={plasma:'#f0d060',buffy:'#ffffff',rbc:'#ef4444'};
    const ctx=cv.getContext('2d');
    ctx.strokeStyle=lc[S.cLayer]||'#fff'; ctx.lineWidth=2; ctx.setLineDash([6,3]);
    ctx.strokeRect(bsx,bsy,p.x-bsx,p.y-bsy); ctx.setLineDash([]);
  };
  cv.onmouseup=e=>{ if(!bdraw)return; bdraw=false; const p=toXY(e.clientX,e.clientY); finishB(p.x,p.y); };
  cv.onmouseleave=()=>{ if(bdraw){bdraw=false;drawBCanvas();} };

  function finishB(x,y){
    const rx=Math.min(bsx,x),ry=Math.min(bsy,y),rw=Math.abs(x-bsx),rh=Math.abs(y-bsy);
    if(rw<5||rh<5)return;
    S.bRegions[S.cLayer]={x:rx,y:ry,w:rw,h:rh};
    drawBCanvas(); toast('✓ '+S.cLayer+' region set');
  }

  let tDraw=false,pinch=false;
  cv.ontouchstart=e=>{
    if(e.touches.length>=2){pinch=true;tDraw=false;bdraw=false;return;}
    pinch=false; e.preventDefault();
    const t=e.touches[0],p=toXY(t.clientX,t.clientY);
    tDraw=true; bdraw=true; bsx=p.x; bsy=p.y;
  };
  cv.ontouchmove=e=>{
    if(pinch||e.touches.length>=2){pinch=true;tDraw=false;bdraw=false;return;}
    if(!tDraw)return; e.preventDefault();
    const t=e.touches[0],p=toXY(t.clientX,t.clientY);
    drawBCanvas();
    const lc={plasma:'#f0d060',buffy:'#ffffff',rbc:'#ef4444'};
    const ctx=cv.getContext('2d');
    ctx.strokeStyle=lc[S.cLayer]||'#fff'; ctx.lineWidth=3; ctx.setLineDash([8,4]);
    ctx.strokeRect(bsx,bsy,p.x-bsx,p.y-bsy); ctx.setLineDash([]);
  };
  cv.ontouchend=e=>{
    if(pinch){pinch=false;return;} if(!tDraw)return;
    tDraw=false; bdraw=false; e.preventDefault();
    const t=e.changedTouches[0],p=toXY(t.clientX,t.clientY);
    finishB(p.x,p.y);
  };
  setupPinchZoom(cv,'b');
}

// ── CORE LAYER MEASUREMENT ──────────────────────────────────────
function measureLayers(cv, regions){
  const H=cv.height;
  const layers={};
  let pH=0,bH=0,rH=0;
  Object.entries(regions).forEach(([name,r])=>{
    const px=getPixels(cv,r.x,r.y,r.w,r.h);
    const rgb=meanRGB(px);
    const hsv=toHSV(rgb.R,rgb.G,rgb.B);
    const lab=toLAB(rgb.R,rgb.G,rgb.B);
    const od =toOD(rgb);
    const ref=HEALTHY[name]||{lab:{L:50,A:0,B:0}};
    const de =dE(lab,ref.lab);
    layers[name]={rgb,hsv,lab,od,de,hpx:r.h,hpct:+((r.h/H)*100).toFixed(1),rawPx:px};
    if(name==='plasma')pH=r.h;
    if(name==='buffy') bH=r.h;
    if(name==='rbc')   rH=r.h;
  });
  const tot=pH+bH+rH||1;
  const hct=+(rH/tot*100).toFixed(1);
  const hb =+(hct/3).toFixed(1);
  const pp =+(pH/tot*100).toFixed(1);
  const bp =+(bH/tot*100).toFixed(1);

  // Clinical enrichment
  const flags={};
  if(layers.plasma){
    flags.icterus=icterusFlag(layers.plasma.lab);
    flags.lipemia=lipemiaFlag(layers.plasma.hsv,layers.plasma.lab);
    flags.plasmaClarity=+(100-layers.plasma.de).toFixed(1);
  }
  if(layers.buffy){
    flags.buffyScore=buffyScore(bp,layers.buffy.hsv.S);
  }
  if(layers.rbc){
    flags.pallorIndex=pallor(layers.rbc.lab.L);
  }

  const age=parseInt(document.getElementById('b-age').value)||25;
  const sex=document.getElementById('b-sex').value||'female';
  const grp=age<12?'child':age<18?(sex==='male'?'teen-m':'teen-f'):(sex==='male'?'adult-m':'adult-f');
  const cls=classify(hb,grp);

  return{layers,hct,hb,pp,bp,cls,flags,pid:document.getElementById('b-pid').value,date:new Date().toISOString().split('T')[0]};
}

function analyseBlood(){
  const cv=document.getElementById('b-cv');
  if(!cv.width){toast('Upload a blood strip image first');return;}
  if(Object.keys(S.bRegions).length===0){toast('Select at least one layer region first');return;}

  if(S.bMode==='sep'||S.bMode==='ref'){
    // Single strip analysis
    S.bloodRes=measureLayers(cv,S.bRegions);
    renderBloodResults(S.bloodRes);
    toast('✓ Blood analysis complete');

  } else if(S.bMode==='dil'){
    // Serial dilution: analyse all loaded slots, build Hb response curve
    S.dilResults=[];
    const slots=Object.keys(S.bImages||{}).map(Number).sort();
    if(slots.length<2){toast('Upload at least 2 dilution images first');return;}
    // For each slot: auto-detect RBC region and measure it
    slots.forEach(slot=>{
      const img=S.bImages[slot];
      const offscreen=document.createElement('canvas');
      const scale=Math.min(1,600/img.width);
      offscreen.width=Math.round(img.width*scale);

        rbc:   {x:4,y:Math.round(h*.57),w:w-8,h:Math.round(h*.43)}
      };
      const res=measureLayers(offscreen,autoReg);
      // dilution factor: 1, 0.5, 0.25, 0.125, 0.0625
      const dilFactor=1/Math.pow(2,slot);
      S.dilResults.push({slot,dilFactor,label:['1/1','1/2','1/4','1/8','1/16'][slot]||('1/'+(slot+1)),...res});
    });
    // Also run on current canvas with manual regions if set
    if(Object.keys(S.bRegions).length>0){
      S.bloodRes=measureLayers(cv,S.bRegions);
    } else {
      S.bloodRes=S.dilResults[0]||null;
    }
    renderDilutionResults();
    toast('✓ Serial dilution analysis complete — '+S.dilResults.length+' strips analysed');

  } else if(S.bMode==='time'){
    // Time points: analyse all loaded strips
    S.timeResults=[];
    const labels=['Week 0','Week 4','Week 8','Week 12'];
    const slots=Object.keys(S.bImages||{}).map(Number).sort();
    if(slots.length<2){toast('Upload at least 2 time-point images first');return;}
    slots.forEach(slot=>{
      const img=S.bImages[slot];
      const offscreen=document.createElement('canvas');
      const scale=Math.min(1,600/img.width);
      offscreen.width=Math.round(img.width*scale);
      offscreen.height=Math.round(img.height*scale);
      offscreen.getContext('2d').drawImage(img,0,0,offscreen.width,offscreen.height);
      const h=offscreen.height,w=offscreen.width;
      const autoReg={
        plasma:{x:4,y:0,w:w-8,h:Math.round(h*.52)},
        buffy: {x:4,y:Math.round(h*.52),w:w-8,h:Math.round(h*.05)},
        rbc:   {x:4,y:Math.round(h*.57),w:w-8,h:Math.round(h*.43)}
      };
      const res=measureLayers(offscreen,autoReg);
      S.timeResults.push({slot,label:labels[slot]||('Week '+(slot*4)),...res});
    });
    S.bloodRes=S.timeResults[0]||null;

  }
}

// ── SIMULATION ────────────────────────────────────────────────
const PROF={
  child:   {nHb:12.5,rda:9,thr:11.0,ab:0.12},
  'teen-m':{nHb:14.5,rda:11,thr:13.0,ab:0.10},
  'teen-f':{nHb:13.0,rda:15,thr:12.0,ab:0.12},
  'adult-m':{nHb:15.5,rda:8,thr:13.0,ab:0.08},
  'adult-f':{nHb:13.5,rda:18,thr:12.0,ab:0.10},
  pregnant:{nHb:12.0,rda:27,thr:11.0,ab:0.20},
  'post-m':{nHb:13.5,rda:8,thr:12.0,ab:0.08},
  elder:   {nHb:13.0,rda:8,thr:12.0,ab:0.09},
};
const FE_PER_HB=3.4; // mg iron per 1 g/dL Hb rise
const DAILY_LOSS=0.9; // mg/day natural iron loss

function runSim(){
  const grp=document.getElementById('s-age').value;
  const p=PROF[grp];
  const hb0=parseFloat(document.getElementById('s-hb').value);
  const fe100=parseFloat(document.getElementById('s-fe').value);
  const sv=parseFloat(document.getElementById('s-sv').value);
  const wks=parseInt(document.getElementById('s-wk').value);
  const vc=parseFloat(document.getElementById('s-vc').value);
  const inh=parseFloat(document.getElementById('s-inh').value);
  const feDay=(fe100/100)*sv; // mg iron consumed per day
  let hb=hb0, totFe=0;
  const rows=[];
  for(let d=0;d<=wks*7;d++){
    if(d%7===0){
      const deficit=Math.max(1,(p.nHb-hb)/p.nHb*3);
      const ar=Math.min(0.35,p.ab*deficit*vc*inh);
      const feAbs=feDay*ar;
      const hbGain=(feAbs-DAILY_LOSS)/FE_PER_HB/7;
      hb=Math.min(p.nHb,Math.max(1,hb+hbGain*7));
      hb=+hb.toFixed(2);
      totFe+=feAbs*7;
      const w=Math.round(d/7);
      const hct=+(hb*3).toFixed(1);
      const cls=classify(hb,grp);
      rows.push({w,hb,hct,feAbs:+(feAbs*7).toFixed(2),ar:+(ar*100).toFixed(1),cls});
    }
  }
  const rec=rows.find(r=>r.hb>=p.thr);
  S.simRes={rows,p,grp,hb0,totFe:+totFe.toFixed(1),rec:rec?.w||null,feDay};
  renderSim();
  toast('✓ Simulation complete');
}

function doExport(fmt,tab){

function dl(content,name,type){
  const a=document.createElement('a');
  a.href=URL.createObjectURL(new Blob([content],{type}));
  a.download=name; a.click();
}

function exportCSV(tab){
  let csv='',name='';
  if(tab==='c'){
    if(!S.cRegions.some(r=>r.rgb)){toast('No colorimetry data');return;}
    csv='Region,Label,R,G,B,Hue,Sat,Val,LAB_L,LAB_A,LAB_B,OD,DeltaE,Conc_mgL\n';
    S.cRegions.filter(r=>r.rgb).forEach(r=>{
      csv+=`R${r.id},${r.label},${r.rgb.R},${r.rgb.G},${r.rgb.B},${r.hsv.H},${r.hsv.S},${r.hsv.V},${r.lab.L},${r.lab.A},${r.lab.B},${r.od},${r.dE},${r.conc??''}\n`;
    });
    name='colorimetry_results.csv';
  } else if(tab==='b'){
    if(!S.bloodRes){toast('No blood data');return;}
    const r=S.bloodRes;
    csv=`Patient,${r.pid}\nDate,${r.date}\nHematocrit%,${r.hct}\nHb_gdL,${r.hb}\nPlasma%,${r.pp}\nBuffy%,${r.bp}\nClassification,${r.cls.l}\n\nLayer,R,G,B,Hue,Sat,LAB_L,LAB_A,OD\n`;
    ['plasma','buffy','rbc'].forEach(n=>{const l=r.layers[n];if(!l)return;csv+=`${n},${l.rgb.R},${l.rgb.G},${l.rgb.B},${l.hsv.H},${l.hsv.S},${l.lab.L},${l.lab.A},${l.od}\n`;});
    name='blood_analysis.csv';
  } else {
    if(!S.simRes){toast('Run simulation first');return;}
    csv='Week,Hb_gdL,Hct_pct,Fe_Abs_mg,Abs_Rate_pct,Status\n';
    S.simRes.rows.forEach(r=>{csv+=`${r.w},${r.hb},${r.hct},${r.feAbs},${r.ar},${r.cls.l}\n`;});
    name='simulation_results.csv';
  }
  dl(csv,name,'text/csv');
  toast('✓ CSV exported');
}

    P(`Analyte: ${document.getElementById('c-analyte').value}`);
    P(`Sample: ${document.getElementById('c-name').value}`);
    P(`Regions: ${S.cRegions.filter(r=>r.rgb).length}`);

    y+=3;T('Region Results',[226,232,240]);
    S.cRegions.filter(r=>r.rgb).forEach(r=>{
      doc.setFillColor(28,36,48);doc.roundedRect(10,y,W-20,16,2,2,'F');
      doc.setFillColor(r.rgb.R,r.rgb.G,r.rgb.B);doc.rect(12,y+3,9,9,'F');
      doc.setTextColor(226,232,240);doc.setFontSize(8);doc.setFont('helvetica','bold');
      doc.text(r.label,25,y+8);
      doc.setFont('helvetica','normal');doc.setTextColor(100,116,139);doc.setFontSize(7);
      doc.text(`RGB(${r.rgb.R},${r.rgb.G},${r.rgb.B}) H:${r.hsv.H}° S:${r.hsv.S}% | LAB A:${r.lab.A} | OD:${r.od} | ΔE:${r.dE}${r.conc!=null?' | '+r.conc+' mg/L':''}`,25,y+13);
      y+=19;
    });
  } else if(tab==='b'){
    T('Blood Strip Analysis Report');L();
    if(S.bloodRes){
      const r=S.bloodRes;
      P(`Patient: ${r.pid||'Unknown'} | Date: ${r.date}`); y+=2;
      const cc={bg:[0,212,160],by:[234,179,8],bo:[249,115,22],br:[239,68,68]};

      doc.setTextColor(...cl);doc.setFontSize(13);doc.setFont('helvetica','bold');
      doc.text(r.cls.l,W/2,y+8,{align:'center'});
      doc.setFontSize(9);doc.text(`Hb: ${r.hb} g/dL  |  Hematocrit: ${r.hct}%`,W/2,y+15,{align:'center'});
      y+=22;L();
      ['plasma','buffy','rbc'].forEach(n=>{const l=r.layers[n];if(!l)return;P(`${n.toUpperCase()}: ${l.hpct}% height | RGB(${l.rgb.R},${l.rgb.G},${l.rgb.B}) | Hue ${l.hsv.H}° | LAB L:${l.lab.L} A:${l.lab.A} | OD:${l.od}`);});
    }
  } else {
    T('Iron Supplementation Simulation Report');L();
    if(S.simRes){
      const r=S.simRes;
      P(`Group: ${document.getElementById('s-age').options[document.getElementById('s-age').selectedIndex].text}`);
      P(`Start Hb: ${r.hb0} g/dL | Target: ${r.p.thr} g/dL | Final Hb: ${r.rows[r.rows.length-1]?.hb} g/dL`);

      y+=3;L();T('Weekly Summary',[226,232,240]);
      r.rows.forEach(row=>{P(`Wk ${String(row.w).padStart(2)} | Hb:${row.hb} g/dL | Hct:${row.hct}% | Fe:${row.feAbs}mg/wk | Abs:${row.ar}% | ${row.cls.l}`,12);});
    }
  }
  y+=8;
  doc.setFillColor(24,30,42);doc.rect(0,285,W,12,'F');
  doc.setTextColor(71,85,105);doc.setFontSize(7);
  doc.text('Research use only — not a clinical diagnostic tool | μFluidic Lab Analyser',W/2,292,{align:'center'});
  doc.save(`${tab}_report.pdf`);
  toast('✓ PDF report saved');
}
